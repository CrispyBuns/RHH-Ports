set -euo pipefail
source /root/buildtools/ports/harbourmasters/lib_build.sh

# Build the Ackbar fork, not upstream HarbourMasters/Shipwright.
# Pin both the repository and branch so autobuilds cannot silently fall back
# to upstream or a release tag.
export REPO_URL=https://github.com/Arrenton/Shipwright.git
export REF=leveled-ackbar
project_clone https://github.com/HarbourMasters/Shipwright.git

# SDL2 from source so the project links against modern SDL2 headers/libraries
# at build time. Do NOT ship libSDL2-2.0.so.0: PortMaster devices provide their
# own SDL2 build, often with device-specific patches.
build_sdl2

# Anchor multiplayer requires SDL2_net, which is not reliably installed on
# target devices, so it is included in the staged runtime libraries.
build_sdl2_net

# Build the same runtime dependency set used by Jeod's SoH port.
build_libzip
# PNG support is required by Shipwright; build zlib first for libpng.
build_zlib
build_libpng
# The Ackbar fork's libultraship CMake requires tinyxml2 at configure time.
build_tinyxml2
build_json
build_bzip2
build_opus
build_opusfile
build_fmt
build_spdlog -DSPDLOG_FMT_EXTERNAL=ON

# Match Jeod's GLES configuration. In this fork's pinned libultraship,
# USE_OPENGLES is explicitly the GLES3 backend (GLSL 300 ES). GLESv2/EGL
# libraries are resolved from the handheld firmware at runtime.
project_configure_and_build GenerateSohOtr \
    -DCMAKE_PREFIX_PATH=/usr/local \
    -DPNG_PNG_INCLUDE_DIR=/usr/local/include/libpng16 \
    -DPNG_LIBRARY=/usr/local/lib/libpng16.so \
    -DUSE_OPENGLES=1 \
    -DBUILD_REMOTE_CONTROL=1 \
    -DCMAKE_BUILD_TYPE=${BUILD_TYPE:-Release} \
    -DCMAKE_EXE_LINKER_FLAGS="-static-libstdc++ -static-libgcc"

# Recursively stage every non-device DT_NEEDED library and verify the full
# dependency closure before the port ZIP is assembled.
stage_libs soh/soh.elf
