set -euo pipefail
source /root/buildtools/ports/harbourmasters/lib_build.sh

project_clone https://github.com/TheRunaway5/Phase-Distorter.git

# Build the checked-in C++ translation without requiring the original assembly
# toolchain. This port deliberately does not bundle ROMs or imported asset packs.
cmake -S "$PROJECT_DIR/cpp" -B "$PROJECT_BUILD_DIR" -G Ninja \
  -DCMAKE_BUILD_TYPE=Release \
  -DEB_PREGENERATED_DIR="$PROJECT_DIR/generated" \
  -DEB_BUILD_TESTS=OFF \
  -DEB_BUILD_FRONTEND=ON \
  -DCMAKE_CXX_FLAGS="-O2 -pipe -mcpu=cortex-a53" \
  -DCMAKE_EXE_LINKER_FLAGS="-Wl,--as-needed"
cmake --build "$PROJECT_BUILD_DIR" --parallel "$(nproc)"

# Give the executable a stable PortMaster-facing name.
mv "$PROJECT_BUILD_DIR/eb_cpp" "$PROJECT_BUILD_DIR/Phase-Distorter"

mkdir -p "$PROJECT_BUILD_DIR/lib"
# Runtime libraries are staged by the shared RHH build environment where needed.
