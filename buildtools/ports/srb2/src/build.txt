#!/usr/bin/env bash
set -euo pipefail

echo "Cloning upstream SRB2 master..."
git clone --depth 1 --single-branch --branch master --recurse-submodules \
    https://github.com/STJr/SRB2.git srb2-src

SRB2_SHA=$(git -C srb2-src rev-parse HEAD)
echo "Building STJr/SRB2 master at ${SRB2_SHA}"

# The PortMaster make variables SDL=1, NOHW=1 and NOUPX=1 are represented
# by this upstream CMake build as follows:
#   SDL=1  -> build the SDL2 target (SRB2SDL2; SDL is required by this target)
#   NOHW=1 -> SRB2_CONFIG_HWRENDER=OFF (software renderer)
#   NOUPX=1 -> no UPX invocation; keep the executable uncompressed
#
# H700 uses Cortex-A53 cores. Tune for the target handheld, never the runner.
# Fast-math is explicitly enabled at the user's request. SRB2's core gameplay
# simulation is predominantly fixed-point; validate gameplay/audio on device.
# LTO is explicitly enabled in compile/link flags and CMake IPO. PGO is off.
CFLAGS_RELEASE="-O3 -DNDEBUG -mcpu=cortex-a53 -ffast-math -fno-math-errno -fno-trapping-math -fno-semantic-interposition -ffunction-sections -fdata-sections -fno-plt -fomit-frame-pointer -flto"
CXXFLAGS_RELEASE="$CFLAGS_RELEASE"
LDFLAGS_RELEASE="-flto -Wl,--gc-sections -Wl,--as-needed -Wl,-O1"

cmake -S srb2-src -B srb2-build -G Ninja \
    -DCMAKE_BUILD_TYPE=Release \
    -DCMAKE_INSTALL_PREFIX="$PWD/srb2-stage" \
    -DCMAKE_C_COMPILER=gcc \
    -DCMAKE_CXX_COMPILER=g++ \
    -DCMAKE_C_FLAGS_RELEASE="$CFLAGS_RELEASE" \
    -DCMAKE_CXX_FLAGS_RELEASE="$CXXFLAGS_RELEASE" \
    -DCMAKE_EXE_LINKER_FLAGS_RELEASE="$LDFLAGS_RELEASE" \
    -DCMAKE_INTERPROCEDURAL_OPTIMIZATION=ON \
    -DSRB2_CONFIG_HWRENDER=OFF \
    -DSRB2_CONFIG_USE_GME=ON \
    -DSRB2_CONFIG_STATIC_STDLIB=OFF \
    -DSRB2_CONFIG_DEV_BUILD=OFF \
    -DSRB2_CONFIG_EXECINFO=ON \
    -DSRB2_CONFIG_PROFILEMODE=OFF

cmake --build srb2-build --target SRB2SDL2 --parallel "$(nproc)"
cmake --install srb2-build

if [[ ! -x srb2-stage/lsdlsrb2 ]]; then
    echo "ERROR: expected installed executable srb2-stage/lsdlsrb2 was not produced." >&2
    find srb2-build/bin srb2-stage -maxdepth 2 -type f -print 2>/dev/null || true
    exit 1
fi

install -m 0755 srb2-stage/lsdlsrb2 ./lsdlsrb2
strip --strip-unneeded ./lsdlsrb2

echo "SRB2 build complete:"
file ./lsdlsrb2
readelf -h ./lsdlsrb2 | grep -E 'Class:|Machine:'
echo "Dynamic dependencies:"
ldd ./lsdlsrb2 || true
