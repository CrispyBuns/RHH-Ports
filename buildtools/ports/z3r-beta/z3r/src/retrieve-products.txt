#!/bin/bash
set -euo pipefail

SRCDIR=$1
DESTDIR=$2

mkdir -p "$DESTDIR/z3r-beta"
cp "$SRCDIR/Z3R-Beta/zelda3" "$DESTDIR/z3r-beta/zelda3"
cp "$SRCDIR/zelda3.user.ini" "$DESTDIR/z3r-beta/zelda3.user.ini"
cp "$SRCDIR/README.md" "$DESTDIR/z3r-beta/README.md"
cp "$SRCDIR/port.json" "$DESTDIR/z3r-beta/port.json"
cp "$SRCDIR/screenshot.jpg" "$DESTDIR/z3r-beta/screenshot.jpg"
cp "$SRCDIR/Z3R-Beta.sh" "$DESTDIR/Z3R-Beta.sh"
chmod +x "$DESTDIR/Z3R-Beta.sh" "$DESTDIR/z3r-beta/zelda3"

echo "Z3R-Beta ARM64 payload staged. Player-generated zelda3_assets.dat is not included."
