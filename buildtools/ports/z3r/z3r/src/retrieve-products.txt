#!/bin/bash
set -euo pipefail

SRCDIR=$1
DESTDIR=$2

mkdir -p "$DESTDIR/z3r"
cp "$SRCDIR/Z3R/zelda3" "$DESTDIR/z3r/zelda3"
cp "$SRCDIR/zelda3.user.ini" "$DESTDIR/z3r/zelda3.user.ini"
cp "$SRCDIR/README.md" "$DESTDIR/z3r/README.md"
cp "$SRCDIR/Z3R.sh" "$DESTDIR/Z3R.sh"
cp "$SRCDIR/port.json" "$DESTDIR/port.json"
cp "$SRCDIR/screenshot.jpg" "$DESTDIR/screenshot.jpg"
chmod +x "$DESTDIR/Z3R.sh" "$DESTDIR/z3r/zelda3"

echo "Z3R ARM64 payload staged. Player-generated zelda3_assets.dat is not included."
